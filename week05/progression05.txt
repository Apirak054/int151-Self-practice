//------------------------------
// 1. filter() + map()
// ใช้กรองข้อมูลที่ตรงเงื่อนไข แล้วแปลงผลลัพธ์
//------------------------------
function studentsPassing(students) {
  return students
    .filter(student => student.score >= 70)        // เลือกเฉพาะคนที่ได้ >= 70
    .map(student => student.name.toUpperCase());   // แปลงชื่อเป็นตัวใหญ่
}

const studentScores = [
  { name: 'Alice', score: 85 },
  { name: 'Bob', score: 92 },
  { name: 'Charlie', score: 68 },
  { name: 'David', score: 55 },
  { name: 'Eve', score: 78 }
];

console.log("filter + map:", studentsPassing(studentScores));

	- filter : เลือกเฉพาะคนที่ score >= 70
	- map : แปลงชื่อเป็นตัวพิมพ์ใหญ่ทั้งหมด

//------------------------------
// 3. รวม Array (concat, spread)
//------------------------------
const num1 = [1,2,3];
const num2 = [4,5,6];

console.log("spread:", [...num1, ...num2]);  // [1,2,3,4,5,6]
console.log("concat:", num1.concat(num2));   // [1,2,3,4,5,6]


//------------------------------
// 4. แปลง Array เป็น String (toString, join)
//------------------------------
console.log("toString:", num1.toString());    // "1,2,3"

console.log("join('.'):", num1.join("."));    // "1.2.3"
console.log("join(','):", num1.join(","));    // "1,2,3"
console.log("join(':'):", num1.join(":"));    // "1:2:3"
console.log("join(';'):", num1.join(";"));    // "1;2;3"
console.log("join('-'):", num1.join("-"));    // "1-2-

	- toString : แยกด้วย , เสมอ
	- join : แยกด้วยตัวคั่นที่กำหนดเอง


//------------------------------
// 5. includes()
// ตรวจสอบว่ามี substring / element หรือไม่
//------------------------------
console.log("tring includes:", "mangoteen".includes("go"));     // true
console.log("string includes:", "mangoteen".includes("Go"));     // false
console.log("array includes:", ["mangoteen","mango"].includes("go"));    // false
console.log("array includes:", ["mangoteen","mango"].includes("mango")); // true
console.log("array includes:", ["mangoteen","mango"].includes("MANGO")); // false

	- ใช้ตรวจสอบว่ามี substring หรือ element อยู่ใน Array/String ไหม
	- case-sensitive (ตัวพิมพ์เล็ก/ใหญ่มีผล)

// เรียง object ตาม name
const students = [
  {id:66500102 , name :"suda" , gpa :2.5},
  {id:66500555 , name :"ada" , gpa :2.8},
  {id:66500589 , name :"pornchai" , gpa :3.25},
  {id:66500102 , name :"Pornsak" , gpa :3.8}
];

const sortedStudents = students.sort((a, b) => {
  const nameA = a.name.toUpperCase(); 
  const nameB = b.name.toUpperCase();
  if (nameA < nameB) return -1;
  if (nameA > nameB) return 1;
  return 0;
});

console.log("sort objects:", sortedStudents);

	- กำหนด compare function เพื่อให้เรียงตามที่เราต้องการ

//------------------------------
// ES Modules 
//------------------------------
//--Named Export--//

// utilities.js
export function echo(msg){ return msg; }
export const GPA = 3.0;

// test.js
import { echo, GPA } from "./utilities.js";
console.log(echo("Hello ES Module"));
console.log(GPA);

    - ต้อง import ด้วย ชื่อเดียวกัน ที่ export ออกไป (จะเปลี่ยนชื่อได้โดยใช้ as)

//--Default Export--//   

// utilities.js
export default function getLength(str){ return str.length; }

// test.js
import getStringLength from "./utilities.js";
console.log(getStringLength("Today"));

    - นำเข้า (import) สามารถตั้งชื่อได้ตามใจ เช่น getLength, strLen, x ก็ได้
