//การสร้าง obj แต่ละแบบ/////////////////////

//------Object Literal //
const phone = { 
    brand : "iphone",
    spec:{cpu:"A18",rom:"256",ram:"8"} 
} 
	- สร้างง่ายที่สุด ใช้ {key: value}
	- เก็บ object ซ้อน ได้ (spec)

//------Constructor Function//
function Ipad(year, color, display) {
    this.year = year;
    this.color = color;
    this.display = display;
}
const Ipad1 = new Ipad("2025","black","13inc")
console.log(Ipad1)

  	- ใช้ function + this
	- เวลาเรียกต้องใช้ new เพื่อสร้าง instance

//-------Class//
class Ipad2 {
    constructor(year, color, display) {
      this.year = year;
      this.color = color;
      this.display = display;
    }
  }
  
  const Ipad3 = new Ipad2("2025", "black", "13inc");
  console.log(Ipad3);  	

    - syntax ใหม่ อ่านง่ายกว่า constructor function
	- รองรับการสืบทอดได้สะดวก

//-------Object.create()//
  const IpadPrototype = {
    info() {
      return `iPad ${this.year}, color: ${this.color}, display: ${this.display}`;
    }
  };
  const Ipad4 = Object.create(IpadPrototype);

Ipad4.year = "2025";
Ipad4.color = "black";
Ipad4.display = "13inc";

console.log(Ipad4);
console.log(Ipad4.info());

    - ใช้สร้าง object ใหม่ โดยอ้างอิง prototype จาก object ที่มีอยู่แล้ว



////////// การเข้าถึง property//////////////////

const per1 = { firstName:"Apirak", lastName:"chaiagsorn", age:20 }
console.log(per1.firstName)   // dot notation
console.log(per1["lastName"]) // bracket notation

	- Dot notation → ใช้ object.key
	- Bracket notation → ใช้ object["key"] (มีประโยชน์ถ้า key ถูกเก็บในตัวแปร)


////////// Object เก็บได้หลายชนิด///////////////
let obj = {
    number: 12,
    string: "age",
    boolean: true,
    undefind: undefined,
    null: null,
    array: [1,2,3],
    obj: { array: [123, 35, 56] },
    objinarr: [{obj:'host'}, {book:123}]
  }

  	- เก็บได้ทั้ง primitive (number, string, boolean, null, undefined)
	- เก็บ array ได้
	- เก็บ object ซ้อนได้
	- เก็บ array ของ object ได้




///////// Methods (ฟังก์ชันใน Object) ////////////
const square = {
  side: 10,
  area(){ return this.side * this.side }
}
console.log(square.area())

	- function ที่อยู่ใน object เรียกว่า method
	- ใช้ this อ้างถึง object ที่มันอยู่




//////// Object Reference ////////////////////
function update(obj){ obj.name = "adam" }
const pet = { name: "eve" }
update(pet)
console.log(pet.name) 

	- object ใน JS เก็บแบบ reference
	- ถ้าแก้ property ในฟังก์ชัน → object ต้นฉบับเปลี่ยนด้วย




/////// Spread //////////////////////////////
const obj1 = {a:1}, obj2={b:2}
const merged = { ...obj1, ...obj2 } 

	- ใช้ ... เพื่อ คัดลอก/รวม property จาก object อื่น
	- ถ้า key ซ้ำ → ตัวหลังจะ overwrite ตัวก่อน





////// Destructuring ///////////////////////

const student = { id:1, name:"Somchai" }
let {id, name} = student
console.log(name) 

	- ใช้ดึงค่าออกจาก object มาเก็บในตัวแปร
	- สามารถเปลี่ยนชื่อได้ เช่น let {name: studentName} = student
